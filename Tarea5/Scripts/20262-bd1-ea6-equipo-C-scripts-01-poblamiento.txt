--INSCRIPCION EVENTO 

INSERT INTO inscripcion_evento (id_cedula, id_evento, fecha_inscripcion)
SELECT
    u.id_cedula,
    e.id_evento,
    CURRENT_DATE - ((u.id_cedula + e.id_evento) % 683)::int
FROM usuario u
CROSS JOIN evento e
WHERE NOT EXISTS (
    SELECT 1
    FROM inscripcion_evento ie
    WHERE ie.id_cedula = u.id_cedula
      AND ie.id_evento = e.id_evento
)
ORDER BY RANDOM()
LIMIT 5000;
------------------------------------------------------------------------

--SOLICITUD_SERVICIO

INSERT INTO solicitud_servicio (id_cedula, id_servicio, fecha_solicitud, estado_solicitud, precio_solicitud)
SELECT 
    u.id_cedula,
    s.id_servicio,
    CURRENT_DATE - ((u.id_cedula + s.id_servicio) % 365)::int,
    CASE ((u.id_cedula + s.id_servicio) % 4)
        WHEN 0 THEN 'Pendiente'
        WHEN 1 THEN 'En proceso'
        WHEN 2 THEN 'Completada'
        ELSE 'Cancelada'
    END,
    (((u.id_cedula + s.id_servicio) % 150) * 1000 + 20000)::numeric
FROM (SELECT id_cedula FROM usuario ORDER BY RANDOM() LIMIT 200) u
CROSS JOIN (SELECT id_servicio FROM servicio ORDER BY RANDOM() LIMIT 50) s
ORDER BY RANDOM()
LIMIT 5000;

-VENTA O Producto_transaccion

INSERT INTO venta (id_cedula_comprador, id_cedula_vendedor, id_producto, fecha_venta, metodo_pago_venta)
SELECT 
    id_comprador,
    id_vendedor,
    id_producto,
    (CURRENT_DATE - floor(random() * 365)::int),
    (ARRAY['Efectivo', 'Tarjeta', 'Transferencia'])[1 + floor(random() * 3)::int]
FROM (
 
    SELECT 
        c.id_cedula AS id_comprador,
        v.id_cedula AS id_vendedor,
        p.id_producto AS id_producto
    FROM (SELECT id_cedula FROM usuario ORDER BY RANDOM() LIMIT 25) c
    CROSS JOIN (SELECT id_cedula FROM usuario ORDER BY RANDOM() LIMIT 25) v
    CROSS JOIN (SELECT id_producto FROM producto ORDER BY RANDOM() LIMIT 10) p
    WHERE c.id_cedula <> v.id_cedula
    ORDER BY random()
    LIMIT 5000
) AS cruce;
